pub trait FiniteField: Clone + PartialEq + Debug + Hash {
    type Characteristic: Into<u64> + Copy + Debug;
    type Wide: Clone;

    fn is_zero(&self) -> bool;
    fn is_one(&self) -> bool;
    fn inv(&self) -> Option<Self>;
    fn extension_degree(&self) -> u32;
    fn characteristic(&self) -> Self::Characteristic;

Trait hierarchy (cont.)

pub trait ConstField: FiniteField + Copy {
    fn zero() -> Self;
    fn one() -> Self;
    fn order() -> u128;      // panics if order > u128::MAX (e.g. GF(2^256))
    fn order_log2() -> u32;  // always safe, even for GF(2^256)
}
